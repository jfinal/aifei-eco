# 配置 undertow。99% 的配置与 jfinal-undertow 相同，学习成本几乎为 0

undertow.port = 80
undertow.host = 0.0.0.0

# 配置 request body 最大值，默认 2M。上传文件大小受此配置控制。以下配置为 10M
undertow.maxBodySize = 10485760

# 配置 css、js、图片等等 web 资源路径
# "src/main/webapp" 表示从普通路径中加载资源， "classpath:static" 表示从 classpath 与 jar 包中的 static 路径下加载资源
# undertow.resourcePath = src/main/webapp, classpath:static

# 配置是否打开资源文件变动监听器，配置为 false 可解决项目资源文件过多所导致的启动过慢问题
# undertow.allowResourceChangeListeners = false

# gzip 压缩开关。nginx 代理时不建议开启
undertow.gzip.enable = false
# 配置压缩级别，默认值 -1。 可配置 1 到 9。 1 拥有最快压缩速度，9 拥有最高压缩率
undertow.gzip.level = -1
# 触发压缩的最小内容长度
undertow.gzip.minLength = 1024
